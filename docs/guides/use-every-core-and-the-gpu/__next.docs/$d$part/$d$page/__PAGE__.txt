/// One core: one number after another.
fn on_one_core(numbers: [Int32]) -> [Int32]
    require numbers.all(n => n >= 1 and n < 100_000)

/// Every core: the rounds of the loop are shared out between the cores,
/// and each round writes only its own place, `out[i]`.
fn on_every_core(numbers: [Int32]) -> [Int32]
    require numbers.all(n => n >= 1 and n < 100_000)
    example on_every_core([6, 7, 27]) == [8, 16, 111]

        out.push(0)
