/// A sensor that fails writes -99. Each such reading is replaced by the
/// mean of that sensor's other readings.
fn repair(t: Tensor<Float>) -> Tensor<Float>
    example repair(tensor.of_rows([[1.0], [-99.0], [3.0]])).items == [1.0, 2.0, 3.0]
{
    const broken = t < -50.0
    const fine = tensor.where(broken, 0.0, t)
    const counts = Tensor<Float>(tensor.where(broken, 0, 1)).sum(axis: 0)
    const means = fine.sum(axis: 0) / counts
    tensor.where(broken, means.reshape([1, t.shape[1]]), t)
}

/// How much each column grows an hour: the slope of the straight line
/// a + b * hour closest to it, by least squares, (AᵀA) x = Aᵀt.

    example (try trends(tensor.of([0.0, 1.0, 2.0]), tensor.of_rows([[1.0], [3.0], [5.0]])))[0].near(2.0)
{
    const a = tensor.stack([tensor.ones<Float>([hours.len()]), hours], axis: 1)
    const fit = try (a.transpose() @ a).solve(a.transpose() @ t)
    fit[1, :]
}

    example two_places(0.4914) == 0.49
