    }
}

/// Refunds orders, within limits a person set, and asks finance first for
/// a large one.
agent Refunder {
    settings AgentSettings(audit: Audit())
    can Payments.refund

    limit Payments.refund {
        amount <= 500
        sum(amount) <= 1_000 per day
    }

    approve Payments.refund when amount > 100 by "finance" within 2.days

    on refund(order: Int, amount: Int) ! PayError {
        try Payments.refund(order, amount)
    }
}
