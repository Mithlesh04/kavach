    this.item = item;
    this.qty = qty;
    this.cents = cents;
  }

  cost() {
    return this.qty * this.cents;
  }

class BadQty extends Error {
  constructor(item) {
    super(`no quantity for ${item}`);
    this.item = item;
  }
}

function total(orders) {
  for (const o of orders) {
    if (o.qty <= 0) {
      throw new BadQty(o.item);
    }
  }
  return orders.reduce((sum, o) => sum + o.cost(), 0);
}
