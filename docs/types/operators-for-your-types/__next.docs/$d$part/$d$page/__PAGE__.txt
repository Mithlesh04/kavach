// Adds every item to `zero`: T needs only +.
fn total<T: Add>(items: [T], zero: T) -> T {
    let sum = zero
    for x in items {
        sum = sum + x
    }
    sum
}

// Turns x negative: T needs only unary -.
fn flip<T: Neg>(x: T) -> T {
    -x
}

// The item that comes last in order, or none for an empty list: T needs only >.
fn largest<T: Comparable>(items: [T]) -> T? {
    let best = items.get(0)
    for x in items {
        if let b = best {
            if x > b {
                best = x
            }
        }
    }
    best
}
