fn member(joins: Sender<Sender<String>>, wants: Int) -> [String] {
    const (send, receive) = Channel<String>(capacity: 4)
    joins.put(send)
    let got: [String] = []
    for _ in 1..=wants {
        match receive.take() {
            Taken.Item(m) => got.push(m),
            Taken.Closed => return got,
        }
    }
    got
}
