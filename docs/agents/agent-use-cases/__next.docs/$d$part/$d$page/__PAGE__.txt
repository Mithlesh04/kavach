    fn read(name: String) -> String ! InboxError

    fn read(self, name: String) -> String ! InboxError uses Fs.read("./inbox") {
        fs.read_text("./inbox/{name}") catch e => match e {
            FsError.Denied(path) => fail InboxError.Denied(reason: "outside ./inbox: {path}"),
            _ => fail InboxError.Missing(name: name),
        }
    }
}
